programa
{
    funcao inicio()
    {
        inteiro num

        escreva("Digite um numero: ")
        leia(num)

        se (num % 2 == 0)
        {
            escreva("O dobro de ", num, " eh ", num * 2, "\n")
        }
        senao
        {
            escreva("O triplo de ", num, " eh ", num * 3, "\n")
        }
    }
}
